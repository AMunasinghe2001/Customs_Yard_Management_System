# Sri Lanka Customs SR Management System - one yard server per PC
# Copy this file beside cus_yard_management.exe, rename it to .env,
# then fill in the values for THIS yard before starting the app.
#
# The profile is any yard code, written in lowercase. It is only the label
# shown in diagnostics; it is not a fixed list. Examples: gl1, gl2, rct,
# nnr, or a new code added later such as kat, bia, or hamb.
#
# The host is the PostgreSQL SERVER PC for this yard, not the client PC.
PG_PROFILE=replace_with_yard_code
PG_HOST=REPLACE_WITH_THIS_YARD_SERVER_IP
PG_PORT=5432
PG_DATABASE=Customs_SR_DB
PG_USER=yard_app
PG_PASSWORD=REPLACE_WITH_THIS_YARD_APP_PASSWORD
PG_SSL=true
PG_SSL_VERIFY=false
PG_CONNECT_TIMEOUT=10
PG_QUERY_TIMEOUT=60
PG_POOL_SIZE=3

# Example for a new yard called KAT:
# PG_PROFILE=kat
# PG_HOST=10.10.71.50
# PG_DATABASE=Customs_SR_DB
# PG_USER=yard_app
# PG_PASSWORD=the-kat-yard-app-password
#
# The app reads these unsuffixed values for the active PC, so no new code or
# new PG_HOST__kat block is required when another yard is created.

# Email settings are shared by the app. Add the real values on the deployed PC.
GMAIL_USER=
GMAIL_APP_PASSWORD=
GMAIL_FROM_NAME=SL Customs
