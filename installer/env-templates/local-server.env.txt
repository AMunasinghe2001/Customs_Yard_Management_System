# ===========================================================================
#  SR Management System — connection settings
#  LOCAL / OFFICE SERVER
#
#  Rename this file to exactly  .env  (no .txt) and put it beside
#  cus_yard_management.exe, normally:
#
#      C:\Program Files\SL Customs\SR Management System\
#
#  ⚠️ Windows Explorer hides known extensions, so "renaming" to .env can
#  quietly leave you with ".env.txt" and the app will still say it cannot
#  reach the database. Turn on View → File name extensions first, or rename
#  it from a Command Prompt:
#
#      ren local-server.env.txt .env
#
#  ⚠️ Installing into Program Files needs administrator rights to save a file
#  there. Open Notepad as administrator, or save it elsewhere and move it in.
# ===========================================================================

# ── Where the database is ──────────────────────────────────────────────────
# The SERVER PC's address on the office network. `localhost` only works on
# the machine that is itself running PostgreSQL — on any other PC it means
# "this PC", and the app will look for a database that is not there.
PG_HOST=10.10.71.15
PG_PORT=5432
PG_DATABASE=Customs_SR_DB

# ── Who the app signs in as ────────────────────────────────────────────────
# ⚠️ yard_app, never postgres. This account is restricted by the per-yard
# row-level security rules; the superuser bypasses all of them, so a copy
# connecting as postgres would show every yard's goods to every office.
PG_USER=yard_app
PG_PASSWORD=PUT_THE_APPLICATION_PASSWORD_HERE

# ── Transport ──────────────────────────────────────────────────────────────
# Off on a trusted office LAN, where the traffic never leaves the building.
# ⚠️ For anything reaching the internet, use the cloud template instead —
# without TLS the password and every row cross the network in the clear.
PG_SSL=false
PG_SSL_VERIFY=false

PG_CONNECT_TIMEOUT=10
PG_QUERY_TIMEOUT=60
