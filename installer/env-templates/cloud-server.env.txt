# ===========================================================================
#  SR Management System — connection settings
#  CLOUD SERVER
#
#  Rename this file to exactly  .env  (no .txt) and put it beside
#  cus_yard_management.exe, normally:
#
#      C:\Program Files\SL Customs\SR Management System\
#
#  ⚠️ Windows Explorer hides known extensions, so "renaming" to .env can
#  quietly leave you with ".env.txt" and the app will still say it cannot
#  reach the database. Turn on View → File name extensions first, or rename
#  it from a Command Prompt:
#
#      ren cloud-server.env.txt .env
#
#  ⚠️ Installing into Program Files needs administrator rights to save a file
#  there. Open Notepad as administrator, or save it elsewhere and move it in.
# ===========================================================================

# ── Where the database is ──────────────────────────────────────────────────
PG_HOST=PUT_THE_CLOUD_HOSTNAME_HERE
PG_PORT=5432
PG_DATABASE=Customs_SR_DB

# ── Who the app signs in as ────────────────────────────────────────────────
# ⚠️ yard_app, never postgres. This account is restricted by the per-yard
# row-level security rules; the superuser bypasses all of them, so a copy
# connecting as postgres would show every yard's goods to every office.
PG_USER=yard_app
PG_PASSWORD=PUT_THE_APPLICATION_PASSWORD_HERE

# ── Transport ──────────────────────────────────────────────────────────────
# ⚠️ TLS is REQUIRED here and must stay on. This connection crosses the
# public internet; without it the application password and every row of
# goods, consignees and valuations travel in the clear, readable by anyone
# on the path. PostgreSQL will happily connect without TLS and say nothing,
# which is exactly why this is stated rather than assumed.
PG_SSL=true

# ⚠️ Verification is what makes TLS mean anything: without it an attacker who
# can answer for the hostname presents their own certificate, the app accepts
# it, and the traffic is encrypted to THEM. Turn this off only if the server
# uses a self-signed certificate, and prefer fixing the certificate.
PG_SSL_VERIFY=true

# A cloud link is slower than a LAN, and a branch on a poor connection should
# not be told the database is down because a query took nine seconds.
PG_CONNECT_TIMEOUT=20
PG_QUERY_TIMEOUT=90
