#  cus_yard_management.exe, normally:
#
#      C:\Program Files\SL Customs\SR Management System\
#
#  ⚠️ Windows Explorer hides known extensions, so "renaming" to .env can
#  quietly leave you with ".env.txt" and the app will still say it cannot
#  reach the database. Turn on View → File name extensions first, or rename

#
#      ren supabase.env.txt .env

# ── Where the database is ──────────────────────────────────────────────────
#
# ⚠️ THE SESSION POOLER, NOT db.<ref>.supabase.co.
#
# The direct host is IPv6-ONLY. From an IPv4 network — which is most of Sri
# Lanka — it fails as "Network is unreachable" or "could not translate host
# name", which reads like a wrong password or a dead server and is neither.
# This cost a whole afternoon once; the pooler is reachable over IPv4.
#
# ⚠️ COPY THIS EXACTLY from your own project — do not assume the region or
# the prefix. Both `aws-0` and `aws-1` exist, and the host carries the region
# your project was created in:
#
#     aws-0-ap-south-1     Mumbai
#     aws-1-ap-southeast-2 Sydney
#
# The wrong one refuses the login. Take the string from:
#     Project Settings → Database → Connection string → Session pooler
PG_HOST=PUT_YOUR_POOLER_HOST_HERE
PG_PORT=5432
PG_DATABASE=postgres

# ── Who the app signs in as ────────────────────────────────────────────────
#
# ⚠️ THE PROJECT REF IS PART OF THE USERNAME on the pooler: `postgres.<ref>`,
# not plain `postgres`. The ref is the random string in your project URL
# (https://<ref>.supabase.co). Without it the pooler cannot tell which
# project you mean and the login fails.
#
# ⚠️ `yard_app`, NEVER `postgres` — the same rule as the office server.
#
# Supabase's own `postgres` role has **rolbypassrls = TRUE**. Measured on a
# live project:
#
#     postgres       super=false  bypassrls=TRUE    <- skips every policy
#     service_role   super=false  bypassrls=TRUE
#     authenticator  super=false  bypassrls=false   (but inherit=false)
#     authenticated  super=false  bypassrls=false
#
# Connecting as `postgres` would skip all 155 per-yard policies and show
# every yard's goods to every office. The app refuses to start on such a
# connection, so this is not advice — it will not run.
#
# ⚠️ Supabase DOES let you create your own role: its `postgres` has
# `rolcreaterole`. Run postgres/91_supabase_yard_app_role.sql to make
# `yard_app`, then put its password below.
#
# ⚠️ The pooler still wants the project ref in the username: `<role>.<ref>`,
# so it is `yard_app.<ref>` here. Measured — a bare role name is refused
# with "FATAL (ENOIDENTIFIER) no tenant identifier provided".
#
# Verify after setup:
#     select rolsuper, rolbypassrls from pg_roles where rolname = current_user;
#       -> both must be FALSE
PG_USER=yard_app.PUT_YOUR_PROJECT_REF_HERE
PG_PASSWORD=PUT_THE_YARD_APP_PASSWORD_HERE

# ── Transport ──────────────────────────────────────────────────────────────
# ⚠️ TLS is REQUIRED here and must stay on. This connection crosses the
# public internet; without it the password and every row of goods,
# consignees and valuations travel in the clear. PostgreSQL will happily
# connect without TLS and say nothing, which is why this is stated.
PG_SSL=true

# ⚠️ Verification is what makes TLS mean anything: without it an attacker who
# can answer for the hostname presents their own certificate, the app accepts
# it, and the traffic is encrypted to THEM.
PG_SSL_VERIFY=true

# ⚠️ SUPABASE USES ITS OWN PRIVATE CA, so verification needs the line below.
#
# The pooler presents  CN=*.pooler.supabase.com,  issued by
# "Supabase Intermediate 2021 CA", under "Supabase Root 2021 CA". That root is
# NOT in Windows' trust store, so `PG_SSL_VERIFY=true` on its own fails the
# handshake with:
#
#     Severity.fatal Bad server certificate.
#
# which names TLS and tells you nothing about the cause. Measured with
# openssl: "Verify return code: 19 (self-signed certificate in certificate
# chain)" without the file, "Verify return code: 0 (ok)" with it.
#
# ⚠️ The answer is NOT to set PG_SSL_VERIFY=false. That keeps the traffic
# encrypted but stops checking WHO it is encrypted to, which is the whole
# protection on a link crossing the public internet.
#
# The file ships beside the .exe (installer/certs/supabase-root-2021.crt).
# A relative name is resolved against the executable's own folder, so this
# works wherever the app is installed. Valid until 2031-04-26.
PG_SSL_ROOT_CERT=supabase-root-2021.crt

# ── Connection budget ──────────────────────────────────────────────────────
#
# ⚠️ ONE connection per client, deliberately.
#
# Every PC holds its own pool, so the server's limit is divided between them:
#
#     clients × (PG_POOL_SIZE + 1 listener) ≤ the plan's limit
#
# Supabase Free allows ~60. At the office default of 3 that is 4 sockets per
# PC and only 15 users fit. At 1 it is 2 sockets, so the current deployment
# — 3 yards + Head Office, about 9 officers — uses 18 of 60, with room to
# spare for a second window or a slow disconnect.
#
# ⚠️ Raise this when the plan changes, NOT when a screen feels slow. A pool
# of 1 serialises a dashboard's panels; that is the trade being made for
# fitting on the free tier. If dashboards feel sluggish, move to Pro and set
# this back to 3 rather than overrunning the limit — the symptom of
# overrunning is `53300: too many connections`, which looks like a server
# fault and is only arithmetic.
PG_POOL_SIZE=1

# ⚠️ The pooler drops connections that sit idle. Retiring ours first means
# the app opens a fresh one on its own schedule instead of discovering a
# dead socket mid-query.
PG_MAX_CONN_AGE=240

# A cloud link is slower than a LAN, and a branch on a poor connection should
# not be told the database is down because a query took nine seconds.
PG_CONNECT_TIMEOUT=20
PG_QUERY_TIMEOUT=90
